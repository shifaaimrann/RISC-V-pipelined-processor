#i type instructions-----------------
addi x1, x0, 10     # x1 = 10
addi x2, x0, 5      # x2 = 5
nop
nop
nop
#R type instructions----------------------------------------
add x3, x1, x2      # x3 = 15
sub x4, x1, x2      # x4 = 5
nop
nop
nop
or x5, x3, x4       # x5 = 15
#memory stores (s-type)------------------------------------
sw x3, 0(x0)        # data_mem[0] = 15
sw x4, 4(x0)        # data_mem[4] = 5
#memory loads (load)-------------------------------------
lw x6, 0(x0)        # load x6 from mem[0]
lw x7, 4(x0)        # load x7 from mem[4]
nop
nop
nop
#load dependency check-------------------------------------
add x8, x6, x7      # x8 = 20
